git config --global user.name "Имя"  - Установить имя пользователя для коммитов
git config --global user.email "почта" - Установить email для коммитов
git init -b main                     - Создать новый локальный репозиторий с веткой main
git status                           - Показать состояние рабочей директории и индекса
git add <файл>                       - Добавить конкретный файл в индекс (staging area)
git add .                            - Добавить все измененные и новые файлы в индекс
git commit -m "Сообщение"            - Зафиксировать изменения с комментарием
git log                              - Показать полную историю коммитов
git log --oneline                    - Показать краткую историю коммитов (хеш + сообщение)
git show <хеш>                       - Показать подробную информацию о конкретном коммите и diff
git branch                           - Показать список локальных веток
git branch -a                        - Показать все ветки (локальные и удаленные)
git checkout -b <имя_ветки>          - Создать новую ветку и сразу переключиться на неё
git switch <имя_ветки>               - Переключиться на существующую ветку (современный способ)
git switch -c <имя_ветки>            - Создать и переключиться (аналог checkout -b)
git branch -d <имя_ветки>            - Удалить ветку (если изменения слиты)
git branch -D <имя_ветки>            - Принудительно удалить ветку
git remote add origin <URL>          - Привязать локальный репозиторий к удаленному
git remote -v                        - Проверить привязанные удаленные репозитории
git push -u origin main              - Отправить ветку main на сервер и связать их
git push --set-upstream origin <ветка> - Отправить новую ветку на сервер
git push                             - Отправить изменения на сервер (если ветка уже связана)
git pull                             - Загрузить изменения с сервера и слить их с локальными
git fetch --prune                    - Обновить информацию о ветках, удаляя несуществующие
